// ReferenceScaler.txt
// ZBrush 2022 ZScript plugin.
// Keeps the Image Plane reference image scaled (and optionally positioned)
// so it tracks the model as you zoom the camera in and out.
//
// Build: ZScript > Load > pick this file. ZBrush compiles it to
// ReferenceScaler.zsc. Copy the .zsc into ZStartup/ZPlugs64 to load it
// automatically. The UI appears under ZPlugin > Reference Scaler.

[IConfig, 2022]
[IShowActions, 0]

// ---------------------------------------------------------------------------
// Image Plane slider paths. These are auto-detected (see ZRSDetect). If your
// Image Plane version names them differently, use "Set Paths" in the plugin
// panel or edit the defaults here.
// ---------------------------------------------------------------------------
[VarDef, zrsSizePath, ""]
[VarDef, zrsHPath, ""]
[VarDef, zrsVPath, ""]

// Anchor state, captured when the reference is lined up with the model.
[VarDef, zrsAnchored, 0]
[VarDef, zrsBaseScale, 0]
[VarDef, zrsBaseSize, 0]
[VarDef, zrsBaseX, 0]
[VarDef, zrsBaseY, 0]
[VarDef, zrsBaseH, 0]
[VarDef, zrsBaseV, 0]

// Live mode.
[VarDef, zrsLive, 0]
[VarDef, zrsInterval, 0.5]

// Scratch values for TransformGet and the update maths.
[VarDef, zrsPx, 0]
[VarDef, zrsPy, 0]
[VarDef, zrsPz, 0]
[VarDef, zrsSx, 0]
[VarDef, zrsSy, 0]
[VarDef, zrsSz, 0]
[VarDef, zrsRx, 0]
[VarDef, zrsRy, 0]
[VarDef, zrsRz, 0]
[VarDef, zrsRatio, 1]
[VarDef, zrsTarget, 0]
[VarDef, zrsMin, 0]
[VarDef, zrsMax, 0]
[VarDef, zrsCx, 0]
[VarDef, zrsCy, 0]
[VarDef, zrsGain, 0]
[VarDef, zrsInput, ""]

[VarDef, zrsDbg, ""]

// ZBrush unloads this script whenever another ZScript runs (Image Plane is a
// ZScript), and reloading resets every VarDef above. Anchor state and paths
// are therefore kept in memory blocks, which live for the whole ZBrush session.
// ZRSState holds 7 floats (4 bytes each): anchored, baseScale, baseSize,
// baseX, baseY, baseH, baseV. ZRSPaths holds 3 strings of 256 bytes each.
[RoutineDef, ZRSSave,
	[If, [MemGetSize, ZRSState] == 0, [MemCreate, ZRSState, 28, 0]]
	[If, [MemGetSize, ZRSPaths] == 0, [MemCreate, ZRSPaths, 768, 0]]
	[MemWrite, ZRSState, zrsAnchored, 0, 0]
	[MemWrite, ZRSState, zrsBaseScale, 0, 4]
	[MemWrite, ZRSState, zrsBaseSize, 0, 8]
	[MemWrite, ZRSState, zrsBaseX, 0, 12]
	[MemWrite, ZRSState, zrsBaseY, 0, 16]
	[MemWrite, ZRSState, zrsBaseH, 0, 20]
	[MemWrite, ZRSState, zrsBaseV, 0, 24]
	[MemWriteString, ZRSPaths, zrsSizePath, 0, 1]
	[MemWriteString, ZRSPaths, zrsHPath, 256, 1]
	[MemWriteString, ZRSPaths, zrsVPath, 512, 1]
]

// Restores state saved by ZRSSave. Every button calls this first.
[RoutineDef, ZRSLoad,
	[If, [MemGetSize, ZRSState],
		[MemRead, ZRSState, zrsAnchored, 0, 0]
		[MemRead, ZRSState, zrsBaseScale, 0, 4]
		[MemRead, ZRSState, zrsBaseSize, 0, 8]
		[MemRead, ZRSState, zrsBaseX, 0, 12]
		[MemRead, ZRSState, zrsBaseY, 0, 16]
		[MemRead, ZRSState, zrsBaseH, 0, 20]
		[MemRead, ZRSState, zrsBaseV, 0, 24]
	]
	[If, [MemGetSize, ZRSPaths],
		[MemReadString, ZRSPaths, zrsSizePath, 0, 1]
		[MemReadString, ZRSPaths, zrsHPath, 256, 1]
		[MemReadString, ZRSPaths, zrsVPath, 512, 1]
	]
	[VarSet, zrsInterval, [IGet, "ZPlugin:Reference Scaler:Interval"]]
]

// Builds a one-line summary of the current state into zrsDbg.
[RoutineDef, ZRSDebugLine,
	[VarSet, zrsDbg, [StrMerge, "RS anchored=", zrsAnchored, " edit=", [IGet, Transform:Edit], " live=", zrsLive]]
	[VarSet, zrsDbg, [StrMerge, zrsDbg, " scale=", zrsSx, " base=", zrsBaseScale, " ratio=", zrsRatio]]
	[If, [StrLength, zrsSizePath],
		[VarSet, zrsDbg, [StrMerge, zrsDbg, " size=", [IGet, #zrsSizePath], " baseSize=", zrsBaseSize]]
		,
		[VarSet, zrsDbg, [StrMerge, zrsDbg, " size=(no slider)"]]
	]
]

// Shows the debug line in the bar at the top of the canvas when Debug is on.
[RoutineDef, ZRSDebug,
	[If, [IGet, "ZPlugin:Reference Scaler:Debug"],
		[RoutineCall, ZRSDebugLine]
		[NoteBar, zrsDbg]
	]
]

// Finds the first existing slider for each Image Plane control.
[RoutineDef, ZRSDetect,
	[VarSet, zrsSizePath, ""]
	[If, [IExists, "Texture:Image Plane:ImgSize"], [VarSet, zrsSizePath, "Texture:Image Plane:ImgSize"]]
	[If, [StrLength, zrsSizePath] == 0, [If, [IExists, "Texture:Image Plane:Img Size"], [VarSet, zrsSizePath, "Texture:Image Plane:Img Size"]]]
	[If, [StrLength, zrsSizePath] == 0, [If, [IExists, "Texture:Image Plane:Image Size"], [VarSet, zrsSizePath, "Texture:Image Plane:Image Size"]]]
	[If, [StrLength, zrsSizePath] == 0, [If, [IExists, "Texture:Image Plane:Size"], [VarSet, zrsSizePath, "Texture:Image Plane:Size"]]]
	[If, [StrLength, zrsSizePath] == 0, [If, [IExists, "Texture:Image Plane:Scale"], [VarSet, zrsSizePath, "Texture:Image Plane:Scale"]]]

	[VarSet, zrsHPath, ""]
	[If, [IExists, "Texture:Image Plane:Shift H"], [VarSet, zrsHPath, "Texture:Image Plane:Shift H"]]
	[If, [StrLength, zrsHPath] == 0, [If, [IExists, "Texture:Image Plane:ShiftH"], [VarSet, zrsHPath, "Texture:Image Plane:ShiftH"]]]

	[VarSet, zrsVPath, ""]
	[If, [IExists, "Texture:Image Plane:Shift V"], [VarSet, zrsVPath, "Texture:Image Plane:Shift V"]]
	[If, [StrLength, zrsVPath] == 0, [If, [IExists, "Texture:Image Plane:ShiftV"], [VarSet, zrsVPath, "Texture:Image Plane:ShiftV"]]]
]

[RoutineDef, ZRSShowPaths,
	[VarSet, zrsInput, "Reference Scaler sliders\n"]
	[If, [StrLength, zrsSizePath],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Size: ", zrsSizePath, "\n"]],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Size: (not found)\n"]]
	]
	[If, [StrLength, zrsHPath],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Shift H: ", zrsHPath, "\n"]],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Shift H: (not found)\n"]]
	]
	[If, [StrLength, zrsVPath],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Shift V: ", zrsVPath]],
		[VarSet, zrsInput, [StrMerge, zrsInput, "Shift V: (not found)"]]
	]
	[Note, zrsInput, , 4]
]

// Records the current model scale/position and Image Plane slider values.
// Line the reference up with the model first, then anchor.
[RoutineDef, ZRSAnchor,
	[VarSet, zrsAnchored, 0]
	[If, [StrLength, zrsSizePath] == 0, [RoutineCall, ZRSDetect]]
	[If, [StrLength, zrsSizePath] == 0,
		[Note, "Reference Scaler: Image Plane size slider not found.\nLoad an image with Texture > Image Plane, or use Set Paths.", , 4]
		,
		[If, [IGet, Transform:Edit] == 0,
			[Note, "Reference Scaler: put the model in Edit mode first.", , 3]
			,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[If, zrsSx <= 0,
				[Note, "Reference Scaler: could not read the model scale.", , 3]
				,
				[VarSet, zrsBaseScale, zrsSx]
				[VarSet, zrsBaseX, zrsPx]
				[VarSet, zrsBaseY, zrsPy]
				[VarSet, zrsBaseSize, [IGet, #zrsSizePath]]
				[If, [StrLength, zrsHPath], [VarSet, zrsBaseH, [IGet, #zrsHPath]]]
				[If, [StrLength, zrsVPath], [VarSet, zrsBaseV, [IGet, #zrsVPath]]]
				[VarSet, zrsAnchored, 1]
			]
		]
	]
	[RoutineCall, ZRSSave]
]

// Rescales (and optionally repositions) the reference to match the model.
[RoutineDef, ZRSUpdate,
	[If, zrsAnchored && [IGet, Transform:Edit],
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[If, zrsSx > 0 && zrsBaseScale > 0,
			[VarSet, zrsRatio, zrsSx / zrsBaseScale]

			[VarSet, zrsTarget, zrsBaseSize * zrsRatio]
			[VarSet, zrsMin, [IGetMin, #zrsSizePath]]
			[VarSet, zrsMax, [IGetMax, #zrsSizePath]]
			[If, zrsTarget < zrsMin, [VarSet, zrsTarget, zrsMin]]
			[If, zrsTarget > zrsMax, [VarSet, zrsTarget, zrsMax]]
			[If, ABS(zrsTarget - [IGet, #zrsSizePath]) > 0.0001,
				[ISet, #zrsSizePath, zrsTarget]
			]

			// Follow Position: the image is assumed to scale about the canvas
			// centre, offset = Gain * slider value (pixels per slider unit).
			// Solving for the offset that keeps the anchored point under the model:
			// slider = base * ratio + ((p - c) - ratio * (p0 - c)) / gain
			[VarSet, zrsGain, [IGet, "ZPlugin:Reference Scaler:Offset Gain"]]
			[If, [IGet, "ZPlugin:Reference Scaler:Follow Position"] && ABS(zrsGain) > 0.0001,
				[VarSet, zrsCx, [IGet, Document:Width] * 0.5]
				[VarSet, zrsCy, [IGet, Document:Height] * 0.5]
				[If, [StrLength, zrsHPath],
					[VarSet, zrsTarget, zrsBaseH * zrsRatio + ((zrsPx - zrsCx) - zrsRatio * (zrsBaseX - zrsCx)) / zrsGain]
					[VarSet, zrsMin, [IGetMin, #zrsHPath]]
					[VarSet, zrsMax, [IGetMax, #zrsHPath]]
					[If, zrsTarget < zrsMin, [VarSet, zrsTarget, zrsMin]]
					[If, zrsTarget > zrsMax, [VarSet, zrsTarget, zrsMax]]
					[If, ABS(zrsTarget - [IGet, #zrsHPath]) > 0.0001, [ISet, #zrsHPath, zrsTarget]]
				]
				[If, [StrLength, zrsVPath],
					[VarSet, zrsTarget, zrsBaseV * zrsRatio + ((zrsPy - zrsCy) - zrsRatio * (zrsBaseY - zrsCy)) / zrsGain]
					[VarSet, zrsMin, [IGetMin, #zrsVPath]]
					[VarSet, zrsMax, [IGetMax, #zrsVPath]]
					[If, zrsTarget < zrsMin, [VarSet, zrsTarget, zrsMin]]
					[If, zrsTarget > zrsMax, [VarSet, zrsTarget, zrsMax]]
					[If, ABS(zrsTarget - [IGet, #zrsVPath]) > 0.0001, [ISet, #zrsVPath, zrsTarget]]
				]
			]
		]
	]
	[RoutineCall, ZRSDebug]
]

// Live loop: wakes on the timer (1) or on left mouse button up (8), i.e. when
// a zoom drag finishes, updates, then goes back to sleep. Another ZScript or
// plugin starting ends the loop (ZBrush only runs one script at a time).
[RoutineDef, ZRSLiveLoop,
	[Sleep, zrsInterval,
		[If, zrsLive,
			[RoutineCall, ZRSUpdate]
			[SleepAgain, zrsInterval, 9]
		]
	, 9]
]
