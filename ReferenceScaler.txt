// ReferenceScaler.txt
// ZBrush 2022 ZScript plugin.
// Scales the Texture > Image Plane reference image with the camera zoom so it
// keeps the same size relative to the model.
//
// Build: ZScript > Load > pick this file. The buttons appear under
// ZPlugin > Reference Scaler b<build>. ZBrush keeps buttons bound to the
// version that first created them until it restarts, so the sub-palette name
// carries the build number: each new build gets fresh buttons without a restart.
//
// Only commands seen working in ZBrush 2022 are used for feedback: [Note, text]
// popups (the note bar is cleared when the script ends, so it shows nothing).
// The anchor is stored in the Anchor Scale / Anchor Size sliders rather than in
// variables, because variables reset whenever ZBrush reloads the script.

[IConfig, 2022]
[IShowActions, 0]

[VarDef, zrsBuild, "build 16"]
[VarDef, zrsLive, 0]
[VarDef, zrsPx, 0]
[VarDef, zrsPy, 0]
[VarDef, zrsPz, 0]
[VarDef, zrsSx, 0]
[VarDef, zrsSy, 0]
[VarDef, zrsSz, 0]
[VarDef, zrsRx, 0]
[VarDef, zrsRy, 0]
[VarDef, zrsRz, 0]
[VarDef, zrsBaseScale, 0]
[VarDef, zrsBaseSize, 0]
[VarDef, zrsBefore, 0]
[VarDef, zrsTarget, 0]
[VarDef, zrsMsg, ""]

// Records the current model scale and Image Size in the anchor sliders and
// leaves a one-line result in zrsMsg.
[RoutineDef, ZRSAnchor,
	[If, [IExists, "Texture:Image Plane:Image Size"] == 0,
		[VarSet, zrsMsg, "Texture > Image Plane > Image Size not found."]
		,
		[If, [IGet, Transform:Edit] == 0,
			[VarSet, zrsMsg, "Put the model in Edit mode first."]
			,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsBaseSize, [IGet, "Texture:Image Plane:Image Size"]]
			[ISet, "ZPlugin:Reference Scaler b16:Anchor Scale", zrsSx]
			[ISet, "ZPlugin:Reference Scaler b16:Anchor Size", zrsBaseSize]
			[VarSet, zrsMsg, [StrMerge, "Anchored: model scale ", zrsSx, ", Image Size ", zrsBaseSize]]
		]
	]
]

// Sets Image Size to anchorSize * (current scale / anchorScale) and leaves a
// result in zrsMsg.
[RoutineDef, ZRSUpdate,
	[VarSet, zrsBaseScale, [IGet, "ZPlugin:Reference Scaler b16:Anchor Scale"]]
	[VarSet, zrsBaseSize, [IGet, "ZPlugin:Reference Scaler b16:Anchor Size"]]
	[If, zrsBaseScale <= 0,
		[VarSet, zrsMsg, "Not anchored: press Anchor first."]
		,
		[If, [IGet, Transform:Edit] == 0,
			[VarSet, zrsMsg, "Put the model in Edit mode first."]
			,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsTarget, zrsBaseSize * zrsSx / zrsBaseScale]
			[VarSet, zrsBefore, [IGet, "Texture:Image Plane:Image Size"]]
			[ISet, "Texture:Image Plane:Image Size", zrsTarget]
			[VarSet, zrsMsg, [StrMerge, "Model scale ", zrsSx, " (anchor ", zrsBaseScale, ")"]]
			[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nImage Size ", zrsBefore, " -> wanted ", zrsTarget]]
			[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nImage Size now ", [IGet, "Texture:Image Plane:Image Size"]]]
		]
	]
]

// Live loop: wakes on the timer (1) or on left mouse button up (8), i.e. when
// a zoom drag finishes, updates, then goes back to sleep. It has no popups,
// since a popup on every update would block zooming. Another ZScript starting
// ends the loop (ZBrush only runs one script at a time).
[RoutineDef, ZRSLiveLoop,
	[Sleep, 0.5,
		[If, zrsLive,
			[RoutineCall, ZRSUpdate]
			[SleepAgain, 0.5, 9]
		]
	, 9]
]

// ---------------------------------------------------------------------------
// Interface
// ---------------------------------------------------------------------------
[ISubPalette, "ZPlugin:Reference Scaler b16"]

[IButton, "ZPlugin:Reference Scaler b16:Anchor",
	"Line the Image Plane reference up with the model, then press to record this as the 1:1 zoom.",
	[RoutineCall, ZRSAnchor]
	[Note, zrsMsg]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b16:Sync",
	"Rescale the reference to the current zoom once.",
	[RoutineCall, ZRSUpdate]
	[Note, zrsMsg]
	, 0, 0.5
]

[ISwitch, "ZPlugin:Reference Scaler b16:Live", 0,
	"Keep rescaling the reference as the camera zooms.",
	[VarSet, zrsLive, 1]
	[RoutineCall, ZRSUpdate]
	[Note, [StrMerge, "Live on.\n", zrsMsg]]
	[RoutineCall, ZRSLiveLoop]
	,
	[VarSet, zrsLive, 0]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b16:Reset",
	"Put Image Size back to its anchored value.",
	[ISet, "Texture:Image Plane:Image Size", [IGet, "ZPlugin:Reference Scaler b16:Anchor Size"]]
	[Note, [StrMerge, "Image Size reset to ", [IGet, "Texture:Image Plane:Image Size"]]]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b16:About",
	"Show which build of the script this button is running.",
	[Note, [StrMerge, "Reference Scaler ", zrsBuild]]
	, 0, 0.5
]

// Anchor storage. Set by Anchor; shown so the anchored values are visible.
[ISlider, "ZPlugin:Reference Scaler b16:Anchor Scale", 0, 0.001, 0, 100000,
	"Model scale recorded by Anchor.",
	, 0, 1
]

[ISlider, "ZPlugin:Reference Scaler b16:Anchor Size", 0, 0.001, 0, 10000,
	"Image Plane Image Size recorded by Anchor.",
	, 0, 1
]
