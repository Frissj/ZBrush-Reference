// ReferenceScaler.txt
// ZBrush 2022+ ZScript plugin.
// Keeps a reference image behind the model scaled and positioned with the
// model as the camera zooms and pans.
//
// ReferenceScaler64.dll renders the reference, scaled and placed for the
// current zoom, into a document-sized PNG. The plugin imports it, selects it
// and paints it in with Color > FillLayer (with the model in Edit mode this
// paints behind it, and X-ray works). The fill drops the model out of Edit
// mode, so after every fill the tool is drawn again and Edit turned back on
// with the saved transform.
//
// Install: keep the ReferenceScaler_Data folder next to this file, then
// ZScript > Load > pick this file. The buttons appear under
// ZPlugin > Reference Scaler b<build>. ZBrush keeps buttons bound to the
// version that first created them until it restarts, so the sub-palette name
// carries the build number: each new build gets fresh buttons without a restart.
//
// ZScript file commands need "!:" in front of an absolute path. Feedback uses
// [Note, text] popups only: the note bar is cleared when the script ends. The
// anchor is stored in sliders, because variables reset whenever ZBrush reloads
// the script.

// Sets the variables for the base layout: the image fitted to the document,
// unscaled (ratio 1, model position = anchor position).
// Warns if the DLL ZBrush has loaded is not the one this build expects:
// once the overlay has run, ZBrush keeps the old DLL until it restarts.
[RoutineDef, ZRSCheckDll,
	[VarSet, zrsTick, [FileExecute, zrsDll, "RSVersion"]]
	[If, zrsTick != zrsDllBuild,
		[Note, [StrMerge, "Wrong helper DLL: it is build ", zrsTick, ", this plugin needs ", zrsDllBuild, ".\nRestart ZBrush, and make sure this file is the DLL being updated:\n", [FileNameResolvePath, zrsDll]]]
	]
]

[RoutineDef, ZRSBase,
	[VarSet, zrsRatio, 1]
	[VarSet, zrsSx, 0]
	[VarSet, zrsPx, 0]
	[VarSet, zrsPy, 0]
	[VarSet, zrsAx, 0]
	[VarSet, zrsAy, 0]
]

// FillLayer drops the model out of Edit mode (it is baked into the canvas and
// painted over). ZRSRestore puts it back: if Edit is off after the fill, it
// draws the current tool at the model's saved position with a canvas drag (in
// Draw mode, which ZBrush returns to when Edit turns off; Transform:Draw is
// pressed only where that path exists),
// turns Edit on and restores the transform saved before the fill (zrsK*).
[RoutineDef, ZRSRestore,
	[VarSet, zrsEditFill, [IGet, Transform:Edit]]
	[If, zrsEditFill == 0,
		[If, [IExists, "Transform:Draw"], [IPress, Transform:Draw]]
		[CanvasClick, zrsKx, zrsKy, zrsKx + 40, zrsKy + 40]
		[IPress, Transform:Edit]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	]
	[VarSet, zrsEditAfter, [IGet, Transform:Edit]]
]

// Renders frame0.png with the DLL for zrsRatio, model position (zrsPx, zrsPy)
// and anchor position (zrsAx, zrsAy), imports it, selects it (an import does
// not make it current; the newest texture is the last one), fills the canvas
// with it and restores the model in Edit mode in front of it. Leaves a result
// in zrsMsg; zrsResult is 1 on success.
// While the overlay is showing, ZRSShow only moves the overlay and leaves the
// canvas cleared.
//
// Clears the canvas layer, imports the frame at zrsFile, selects it and fills
// the canvas with it. The fill covers the model until ZRSRestore draws it
// again. Without the clear, FillLayer keeps the depth of the model it baked
// into the canvas, and the redrawn model fights with that copy (stripes of
// fill colour on the model, and clipping when it is smaller than the copy).
[RoutineDef, ZRSFillOnly,
	[VarSet, zrsCleared, [IExists, "Layer:Clear"]]
	[If, zrsCleared, [IPress, "Layer:Clear"]]
	[FileNameSetNext, [StrMerge, "!:", zrsFile]]
	[IPress, Texture:Import]
	[ISet, "Texture:Item Info", [IGetMax, "Texture:Item Info"]]
	[IPress, Color:FillLayer]
]

// Fills the canvas with the frame at zrsFile and puts the model back in Edit
// mode in front of it.
[RoutineDef, ZRSFill,
	[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSFillOnly]
	[RoutineCall, ZRSRestore]
]

[RoutineDef, ZRSShow,
	[VarSet, zrsResult, 0]
	[If, [IGet, Transform:Edit] == 0,
		[VarSet, zrsMsg, "Put the model in Edit mode first."]
		,
		[VarSet, zrsOverlayOn, [FileExecute, zrsDll, "RSOverlayInfo", "4"]]
		[If, zrsOverlayOn == 1,
			[VarSet, zrsResult, 1]
			,
			[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " ", zrsRatio, " ", zrsPx]]
			[VarSet, zrsArgs, [StrMerge, zrsArgs, " ", zrsPy, " ", zrsAx, " ", zrsAy]]
			[VarSet, zrsArgs, [StrMerge, zrsArgs, " frame0.png"]]
			[VarSet, zrsResult, [FileExecute, zrsDll, "RSRender", zrsArgs]]
			[If, zrsResult == 1,
				[VarSet, zrsFile, [FileNameResolvePath, "ReferenceScaler_Data\frame0.png"]]
				[RoutineCall, ZRSFill]
			]
		]
		[If, zrsResult != 1,
			[VarSet, zrsMsg, [StrMerge, "Render failed (", zrsResult, "). -3 means no image: press Load Image.\nArgs: ", zrsArgs]]
			,
			[VarSet, zrsLastRatio, zrsRatio]
			[VarSet, zrsLastPx, zrsPx]
			[VarSet, zrsLastPy, zrsPy]
			[VarSet, zrsArgs, [StrMerge, zrsRatio, " ", zrsPx, " ", zrsPy, " ", zrsAx, " ", zrsAy]]
			[VarSet, zrsSetResult, [FileExecute, zrsDll, "RSOverlaySet", [StrMerge, zrsArgs, " ", zrsSx]]]
			[If, zrsSetResult == 3, [VarSet, zrsLastRatio, -1]]
			[VarSet, zrsMsg, [StrMerge, "Shown at ratio ", zrsRatio, ", model at ", zrsPx, ", ", zrsPy]]
			[If, zrsOverlayOn == 1,
				[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nOverlay only. Capture ", [FileExecute, zrsDll, "RSOverlayInfo", "20"], " (1 = GPU), ", [FileExecute, zrsDll, "RSOverlayInfo", "21"], " fps"]]
				[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nZBrush memory: locked ", [FileExecute, zrsDll, "RSOverlayInfo", "22"], ", candidates ", [FileExecute, zrsDll, "RSOverlayInfo", "23"], " ", [FileExecute, zrsDll, "RSOverlayInfo", "24"]]]
				[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nLast drag off by ", [FileExecute, zrsDll, "RSOverlayInfo", "12"], " px, scale x", [FileExecute, zrsDll, "RSOverlayInfo", "13"]]]
				[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nLearned: pan ", [FileExecute, zrsDll, "RSOverlayInfo", "6"], ", zoom ", [FileExecute, zrsDll, "RSOverlayInfo", "7"], " per 1000 px"]]
				[VarSet, zrsMsg, [StrMerge, zrsMsg, ", pivot ", [FileExecute, zrsDll, "RSOverlayInfo", "9"], " (", [FileExecute, zrsDll, "RSOverlayInfo", "10"], " moves, ", [FileExecute, zrsDll, "RSOverlayInfo", "11"], " zooms)"]]
				,
				[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nEdit after fill ", zrsEditFill, ", after restore ", zrsEditAfter]]
			]
		]
	]
]

// Records the current view as the anchor, in the sliders and in the DLL's
// anchor.txt (which outlives reloads and new builds).

	[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor Scale", zrsSx]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor X", zrsPx]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor Y", zrsPy]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot X", zrsRx]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot Y", zrsRy]
	[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot Z", zrsRz]
	[VarSet, zrsArgs, [StrMerge, zrsSx, " ", zrsPx, " ", zrsPy, " ", zrsRx, " ", zrsRy]]
	[FileExecute, zrsDll, "RSAnchorSave", [StrMerge, zrsArgs, " ", zrsRz]]
	[FileExecute, zrsDll, "RSOverlayResetOffset"]
	[VarSet, zrsMsg, [StrMerge, "Anchored: model scale ", zrsSx, ", position ", zrsPx, ", ", zrsPy]]
]

// Fills empty anchor sliders (a new build or a reload) from anchor.txt.
[RoutineDef, ZRSLoadAnchor,
	[If, [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"] <= 0,
		[VarSet, zrsTick, [FileExecute, zrsDll, "RSAnchorGet", "0"]]
		[If, zrsTick > 0,
			[ISet, "ZPlugin:Reference Scaler b65:Anchor Scale", zrsTick]
			[ISet, "ZPlugin:Reference Scaler b65:Anchor X", [FileExecute, zrsDll, "RSAnchorGet", "1"]]
			[ISet, "ZPlugin:Reference Scaler b65:Anchor Y", [FileExecute, zrsDll, "RSAnchorGet", "2"]]
			[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot X", [FileExecute, zrsDll, "RSAnchorGet", "3"]]
			[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot Y", [FileExecute, zrsDll, "RSAnchorGet", "4"]]
			[ISet, "ZPlugin:Reference Scaler b65:Anchor Rot Z", [FileExecute, zrsDll, "RSAnchorGet", "5"]]
		]
	]
]

// Works out the ratio and positions from the current model, the anchor
// sliders and the rotation offset, then shows the frame. Leaves a result in
// zrsMsg.
[RoutineDef, ZRSUpdate,
	[RoutineCall, ZRSLoadAnchor]
	[If, [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"] <= 0,
		[VarSet, zrsMsg, "Not anchored: line the model up with the image and press Anchor first."]
		[VarSet, zrsResult, 0]
		,
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsRatio, zrsSx / [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]
		[VarSet, zrsAx, [IGet, "ZPlugin:Reference Scaler b65:Anchor X"]]
		[VarSet, zrsAy, [IGet, "ZPlugin:Reference Scaler b65:Anchor Y"]]
		[RoutineCall, ZRSShow]
	]
]

// Live loop: wakes on the timer (1) or on left mouse button up (8), i.e. when
// a zoom drag finishes, and redraws only if the zoom or position changed.
// A rotation drag must not move the reference: the DLL keeps whatever it
// moves the model as an offset. When the view's rotation is back at the
// anchored rotation (for example from the navigation head), the offset is
// cleared and the reference snaps back to its anchor.
// Popups appear only when a redraw fails, or on every wake with Diag on. Another plugin's button pressed by
// you (not by this script) ends the loop.
[RoutineDef, ZRSLiveLoop,
	[Sleep, 0.5,
		[If, zrsLive,
			[VarSet, zrsTick, zrsTick + 1]
			[If, [IGet, Transform:Edit],
				[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
				[VarSet, zrsRatio, zrsSx / [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]
				[VarSet, zrsKind, [FileExecute, zrsDll, "RSOverlayInfo", "19"]]
				[VarSet, zrsArgs, [StrMerge, "live tick ", zrsTick, " kind ", zrsKind, " scale ", zrsSx]]
				[FileExecute, zrsDll, "RSLog", [StrMerge, zrsArgs, " pos ", zrsPx, " ", zrsPy, " z ", zrsPz]]
				[VarSet, zrsForce, 0]
				[VarSet, zrsTurn, ABS(zrsRx - [IGet, "ZPlugin:Reference Scaler b65:Anchor Rot X"]) + ABS(zrsRy - [IGet, "ZPlugin:Reference Scaler b65:Anchor Rot Y"])]
				[VarSet, zrsTurn, zrsTurn + ABS(zrsRz - [IGet, "ZPlugin:Reference Scaler b65:Anchor Rot Z"])]
				[If, zrsTurn < 1, [VarSet, zrsForce, [FileExecute, zrsDll, "RSOverlayResetOffset"]]]
				[VarSet, zrsTry, [IGet, "ZPlugin:Reference Scaler b65:Depth Factor"] * zrsSx]
				[If, [IGet, "ZPlugin:Reference Scaler b65:Depth Factor"] != 0 && ABS(zrsPz - zrsTry) > ABS(zrsTry) * 0.05,
					[TransformSet, zrsPx, zrsPy, zrsTry, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
				]
				[If, zrsForce || ABS(zrsRatio - zrsLastRatio) > 0.002 || ABS(zrsPx - zrsLastPx) > 0.5 || ABS(zrsPy - zrsLastPy) > 0.5,
					[RoutineCall, ZRSUpdate]
					[If, [IGet, "ZPlugin:Reference Scaler b65:Diag"] || zrsResult != 1, [Note, [StrMerge, "Live tick ", zrsTick, ": redrew, result ", zrsResult, "\n", zrsMsg], , 1]]
					,
					[If, [IGet, "ZPlugin:Reference Scaler b65:Diag"], [Note, [StrMerge, "Live tick ", zrsTick, ": no change (ratio ", zrsRatio, ")"], , 0.5]]
				]
				,
				[If, [IGet, "ZPlugin:Reference Scaler b65:Diag"], [Note, [StrMerge, "Live tick ", zrsTick, ": model not in Edit mode"], , 0.5]]
			]
			[SleepAgain, 0.5, 9]
		]
	, 9]
]

// Overlay (stage 1 of the live reference): a click-through window drawn by
// the DLL over the canvas. The DLL finds the canvas on screen itself, by
// matching the calibration pattern (which the canvas must be showing) against
// a capture of ZBrush's window. It starts with the last rendered frame's zoom
// and position.
[RoutineDef, ZRSOverlayOn,
	[IUpdate, 2, 1]
	[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " ", [IGet, "ZPlugin:Reference Scaler b65:Overlay Opacity"] * 2.55]]
	[VarSet, zrsResult, [FileExecute, zrsDll, "RSOverlayStart", zrsArgs]]
	[VarSet, zrsOx, [FileExecute, zrsDll, "RSOverlayInfo", "0"]]
	[VarSet, zrsOy, [FileExecute, zrsDll, "RSOverlayInfo", "1"]]
	[VarSet, zrsZoom, [FileExecute, zrsDll, "RSOverlayInfo", "2"] / 1000]
	[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "3"]]
]

// The key colour as the document background (Document > Back, gradient Range
// 0) on a cleared canvas: unlike filled pixels it has no depth, so the model
// can never go behind it. The user's background is saved in two sliders and
// put back by Overlay Off.
[RoutineDef, ZRSKeyBackground,
	[If, [IExists, "Document:Back"],
		[VarSet, zrsTick, [IGet, "Document:Back"]]
		[If, zrsTick != 16711935,
			[ISet, "ZPlugin:Reference Scaler b65:Saved Back", zrsTick]
			[If, [IExists, "Document:Range"], [ISet, "ZPlugin:Reference Scaler b65:Saved Range", [IGet, "Document:Range"]]]
		]
		[IColorSet, 255, 0, 255]
		[IPress, "Document:Back"]
		[If, [IExists, "Document:Range"], [ISet, "Document:Range", 0]]
		[IColorSet, 255, 255, 255]
	]
	[VarSet, zrsCleared, [IExists, "Layer:Clear"]]
	[If, zrsCleared, [IPress, "Layer:Clear"]]
]

[RoutineDef, ZRSRestoreBackground,
	[VarSet, zrsTick, [IGet, "ZPlugin:Reference Scaler b65:Saved Back"]]
	[If, zrsTick > 0 && [IExists, "Document:Back"],
		[VarSet, zrsKx, INT(zrsTick / 65536)]
		[VarSet, zrsKy, INT((zrsTick - zrsKx * 65536) / 256)]
		[IColorSet, zrsKx, zrsKy, zrsTick - zrsKx * 65536 - zrsKy * 256]
		[IPress, "Document:Back"]
		[IColorSet, 255, 255, 255]
		[If, [IExists, "Document:Range"], [ISet, "Document:Range", [IGet, "ZPlugin:Reference Scaler b65:Saved Range"]]]
	]
]

// Finds where ZBrush keeps the model's view in memory, so the DLL can follow
// zooms and pans exactly: reports the current view, sets three test views
// (moved and scaled a little) and a rotated one (which rules out rotation
// matrix entries that equal the scale in the current view) and reports each,
// then puts the view back and reports that. zrsResult is 1 when the DLL has locked on.
[RoutineDef, ZRSMemProbe,
	[VarSet, zrsResult, 0]
	[If, [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"] > 0,
		[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
		[VarSet, zrsArgs, [StrMerge, "start ", zrsKsx, " ", zrsKx, " ", zrsKy, " ", [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]]
		[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		[Loop, 3,
			[VarSet, zrsTick, zrsTick + 1]
			[TransformSet, zrsKx + 11 * zrsTick, zrsKy + 7 * zrsTick, zrsKz, zrsKsx * (1 + 0.04 * zrsTick), zrsKsy * (1 + 0.04 * zrsTick), zrsKsz * (1 + 0.04 * zrsTick), zrsKrx, zrsKry, zrsKrz]
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsArgs, [StrMerge, "next ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]]
			[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx + 15, zrsKry + 25, zrsKrz]
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsArgs, [StrMerge, "next ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]]
		[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsArgs, [StrMerge, "last ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]]
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSMemProbe", zrsArgs]]
	]
]

// Logs ZBrush's canvas depth (PixolPick component 1) and colour (0) on a 5 x 5
// grid over the document, labelled with zrsMsg2, to nav_log.txt: shows
// whether empty canvas has depth left in it that could hide the model.
[RoutineDef, ZRSDepthProbe,
	[VarSet, zrsN, 0]
	[Loop, 25,
		[VarSet, zrsTry, [IGet, Document:Width] * (0.05 + 0.225 * (zrsN - INT(zrsN / 5) * 5))]
		[VarSet, zrsMult, [IGet, Document:Height] * (0.05 + 0.225 * INT(zrsN / 5))]
		[VarSet, zrsTry, INT(zrsTry)]
		[VarSet, zrsMult, INT(zrsMult)]
		[VarSet, zrsArgs, [StrMerge, zrsMsg2, " at ", zrsTry, " ", zrsMult, " depth "]]
		[FileExecute, zrsDll, "RSLog", [StrMerge, zrsArgs, [PixolPick, 1, zrsTry, zrsMult], " colour ", [PixolPick, 0, zrsTry, zrsMult]]]
		[VarSet, zrsN, zrsN + 1]
	]
]

// Counts in zrsCount how many points of a 30 x 30 grid over the document
// ZBrush has drawn something other than the key colour at (the model).
[RoutineDef, ZRSCountModel,
	[VarSet, zrsCount, 0]
	[VarSet, zrsN, 0]
	[Loop, 900,
		[VarSet, zrsTry, [IGet, Document:Width] * (0.5 + zrsN - INT(zrsN / 30) * 30) / 30]
		[VarSet, zrsMult, [IGet, Document:Height] * (0.5 + INT(zrsN / 30)) / 30]
		[If, [PixolPick, 0, zrsTry, zrsMult] != 16711935, [VarSet, zrsCount, zrsCount + 1]]
		[VarSet, zrsN, zrsN + 1]
	]
]

// ZBrush stops drawing what is behind its back clipping plane, and a model
// at the canvas plane (depth 0) is cut off when zoomed out. The ZBrush fix is
// to bring the model forward in depth. This shrinks the view to 30% (where
// the cut-off shows), counts how much of the model ZBrush draws with it 5x
// its scale in front of and behind its depth, keeps whichever shows more,
// and restores the zoom. Live keeps the depth in proportion (Depth Factor).
[RoutineDef, ZRSDepthFix,
	[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[VarSet, zrsBestZ, zrsKz]
	[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[VarSet, zrsBestCount, zrsCount]
	[VarSet, zrsMemFound, zrsCount]
	[TransformSet, zrsKx, zrsKy, zrsKz - 5 * zrsKsx, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[VarSet, zrsTurn, zrsCount]
	[If, zrsCount > zrsBestCount + 2, [VarSet, zrsBestCount, zrsCount] [VarSet, zrsBestZ, zrsKz - 5 * zrsKsx]]
	[TransformSet, zrsKx, zrsKy, zrsKz + 5 * zrsKsx, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[If, zrsCount > zrsBestCount + 2, [VarSet, zrsBestCount, zrsCount] [VarSet, zrsBestZ, zrsKz + 5 * zrsKsx]]
	[VarSet, zrsArgs, [StrMerge, "depth fix: z ", zrsKz, " shows ", zrsMemFound, ", z-5s shows ", zrsTurn, ", z+5s shows ", zrsCount]]
	[FileExecute, zrsDll, "RSLog", [StrMerge, zrsArgs, ", using ", zrsBestZ]]
	[TransformSet, zrsKx, zrsKy, zrsBestZ, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[If, zrsBestZ != zrsKz,
		[ISet, "ZPlugin:Reference Scaler b65:Depth Factor", zrsBestZ / zrsKsx]
		,
		[ISet, "ZPlugin:Reference Scaler b65:Depth Factor", 0]
	]
]

// Overlay: works out the layout for the current camera (rendering it to
// frame0.png also saves the layout for the DLL), fills the canvas with the
// calibration pattern, which also hides the model, waits a second so ZBrush
// repaints the screen, lets the DLL find the pattern on screen, then fills the
// canvas with the key colour magenta (or with the reference if that failed)
// and puts the model back. The overlay shows the reference wherever it sees
// magenta, so the model stays in front.
// Every fill uses frame0.png: importing a name that already exists replaces
// that texture in place, so a single name keeps it the last texture.
[IButton, "ZPlugin:Reference Scaler b65:Overlay",
	"Show the reference in a click-through window over the canvas, behind the model. Press again to find the canvas again.",
	[RoutineCall, ZRSCheckDll]
	[FileExecute, zrsDll, "RSOverlayStop"]
	[If, [IGet, Transform:Edit] == 0,
		[Note, "Put the model in Edit mode first."]
		,
		[RoutineCall, ZRSLoadAnchor]
		[VarSet, zrsAutoAnchored, 0]
		[If, [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"] <= 0,
			[RoutineCall, ZRSAnchorNow]
			[VarSet, zrsAutoAnchored, 1]
		]
		[If, [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"] > 0,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsRatio, zrsSx / [IGet, "ZPlugin:Reference Scaler b65:Anchor Scale"]]
			[VarSet, zrsAx, [IGet, "ZPlugin:Reference Scaler b65:Anchor X"]]
			[VarSet, zrsAy, [IGet, "ZPlugin:Reference Scaler b65:Anchor Y"]]
			,
			[RoutineCall, ZRSBase]
		]
		[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " ", zrsRatio, " ", zrsPx]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " ", zrsPy, " ", zrsAx, " ", zrsAy]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " frame0.png"]]
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSRender", zrsArgs]]
		[If, zrsResult != 1,
			[Note, [StrMerge, "Render failed (", zrsResult, "). -3 means no image: press Load Image."]]
			,
			[FileExecute, zrsDll, "RSPattern", [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " frame0.png"]]
			[VarSet, zrsFile, [FileNameResolvePath, "ReferenceScaler_Data\frame0.png"]]
			[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
			[VarSet, zrsMsg2, "before overlay"]
			[RoutineCall, ZRSDepthProbe]
			[RoutineCall, ZRSFillOnly]
			[VarSet, zrsMsg2, "pattern fill"]
			[RoutineCall, ZRSDepthProbe]
			[Sleep, 1, [VarSet, zrsTick, 0]]
			[RoutineCall, ZRSOverlayOn]
			[If, zrsResult > 0,
				[RoutineCall, ZRSKeyBackground]
				,
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSRender", zrsArgs]]
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "3"]]
				[RoutineCall, ZRSFillOnly]
			]
			[RoutineCall, ZRSRestore]
			[VarSet, zrsLastRatio, zrsRatio]
			[VarSet, zrsLastPx, zrsPx]
			[VarSet, zrsLastPy, zrsPy]
			[If, zrsResult > 0,
				[VarSet, zrsArgs, [StrMerge, "Overlay on: found the canvas at ", zrsOx, ", ", zrsOy, ", ", zrsZoom, " screen px per document px (match ", zrsTick, "%)"]]
				[VarSet, zrsResult, [FileExecute, zrsDll, "RSOverlayInfo", "5"]]
				[If, zrsResult == 1,
					[VarSet, zrsMsg, "Model in front: keying on (1)."]
					,
					[VarSet, zrsMsg, [StrMerge, "Keying unavailable (", zrsResult, "): 0 = this Windows cannot exclude the overlay from capture, -1 = exclusion did not work. The overlay covers the model."]]
				]
				[Sleep, 0.5, [VarSet, zrsTick, 0]]
				[VarSet, zrsMsg2, "overlay ready"]
				[RoutineCall, ZRSDepthProbe]
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "32"]]
				[If, zrsTick < 5,
					[FileExecute, zrsDll, "RSBlank", [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " frame0.png 255 0 255"]]
					[VarSet, zrsFile, [FileNameResolvePath, "ReferenceScaler_Data\frame0.png"]]
					[RoutineCall, ZRSFill]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nBackground colour did not turn magenta (", zrsTick, "%): using a fill, the model may clip."]]
				]
				[VarSet, zrsTick, 0]
				[RoutineCall, ZRSMemProbe]
				[VarSet, zrsSetResult, zrsResult]
				[RoutineCall, ZRSDepthFix]
				[VarSet, zrsResult, zrsSetResult]
				[If, zrsResult == 1,
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nFound ZBrush's view in memory: zoom and pan follow exactly."]]
					[If, [IGet, "ZPlugin:Reference Scaler b65:Depth Factor"] != 0,
						[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nMoved the model off ZBrush's back clipping plane (depth ", zrsBestZ, ")."]]
						,
						[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nDepth unchanged: the test drew ", zrsMemFound, " / ", zrsTurn, " / ", zrsCount, " points."]]
					]
					,
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nZBrush's view not found in memory (", zrsResult, "), candidates ", [FileExecute, zrsDll, "RSOverlayInfo", "23"], " ", [FileExecute, zrsDll, "RSOverlayInfo", "24"], " ", [FileExecute, zrsDll, "RSOverlayInfo", "25"]]]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nScan: ", [FileExecute, zrsDll, "RSOverlayInfo", "27"], " MB, x matches ", [FileExecute, zrsDll, "RSOverlayInfo", "28"], ", pairs ", [FileExecute, zrsDll, "RSOverlayInfo", "29"]]]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, ", scales ", [FileExecute, zrsDll, "RSOverlayInfo", "30"], ", ", [FileExecute, zrsDll, "RSOverlayInfo", "31"], " ms. View ", zrsKsx, " ", zrsKx, " ", zrsKy]]
				]
				[If, zrsCleared == 0, [VarSet, zrsMsg, [StrMerge, zrsMsg, "\nLayer:Clear not found, the model may show stripes."]]]
				[If, zrsAutoAnchored, [VarSet, zrsMsg, [StrMerge, zrsMsg, "\nNo anchor was set: anchored at the current view."]]]
				[Note, [StrMerge, zrsArgs, "\n", zrsMsg]]
				,
				[VarSet, zrsMsg, "-3: frame0.png missing.\n-4: screen capture failed.\n-5: the pattern was not found on screen."]
				[VarSet, zrsArgs, [StrMerge, "Overlay failed (", zrsResult, "). Best match ", zrsTick, "% at ", zrsOx, ", ", zrsOy]]
				[Note, [StrMerge, zrsArgs, " scale ", zrsZoom, "\n", zrsMsg]]
				[Note, [StrMerge, "What the plugin saw is saved as\n", [FileNameResolvePath, "ReferenceScaler_Data\capture.png"], "\nPlease send that file."]]
			]
		]
	]
	, 0, 0.5
]
