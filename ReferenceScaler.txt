// ReferenceScaler.txt
// ZBrush 2022 ZScript plugin.
// Scales the Texture > Image Plane reference image with the camera zoom so it
// keeps the same size relative to the model.
//
// Build: ZScript > Load > pick this file. The buttons appear under
// ZPlugin > Reference Scaler b<build>. ZBrush keeps buttons bound to the
// version that first created them until it restarts, so the sub-palette name
// carries the build number: each new build gets fresh buttons without a restart.

[IConfig, 2022]
[IShowActions, 0]

// Shown by About and in every note-bar message, to confirm which version a
// button is running.
[VarDef, zrsBuild, "build 15"]

// Anchor state, captured when the reference is lined up with the model.
[VarDef, zrsAnchored, 0]
[VarDef, zrsBaseScale, 0]
[VarDef, zrsBaseSize, 0]

// Live mode.
[VarDef, zrsLive, 0]

// Scratch values.
[VarDef, zrsPx, 0]
[VarDef, zrsPy, 0]
[VarDef, zrsPz, 0]
[VarDef, zrsSx, 0]
[VarDef, zrsSy, 0]
[VarDef, zrsSz, 0]
[VarDef, zrsRx, 0]
[VarDef, zrsRy, 0]
[VarDef, zrsRz, 0]
[VarDef, zrsRatio, 1]
[VarDef, zrsBefore, 0]
[VarDef, zrsTarget, 0]
[VarDef, zrsMsg, ""]

// ZBrush unloads this script whenever another ZScript runs (Image Plane is a
// ZScript), and reloading resets every VarDef above. The anchor is kept in a
// memory block, which lives for the whole ZBrush session.
// ZRSState holds 3 floats: anchored, baseScale, baseSize.
[RoutineDef, ZRSSave,
	[If, [MemGetSize, ZRSState] == 0, [MemCreate, ZRSState, 12, 0]]
	[MemWrite, ZRSState, zrsAnchored, 0, 0]
	[MemWrite, ZRSState, zrsBaseScale, 0, 4]
	[MemWrite, ZRSState, zrsBaseSize, 0, 8]
]

[RoutineDef, ZRSLoad,
	[If, [MemGetSize, ZRSState],
		[MemRead, ZRSState, zrsAnchored, 0, 0]
		[MemRead, ZRSState, zrsBaseScale, 0, 4]
		[MemRead, ZRSState, zrsBaseSize, 0, 8]
	]
]

// Records the current model scale and Image Size. Line the reference up with
// the model first.
[RoutineDef, ZRSAnchor,
	[VarSet, zrsAnchored, 0]
	[If, [IExists, "Texture:Image Plane:Image Size"] == 0,
		[Note, "Reference Scaler: Texture > Image Plane > Image Size not found."]
		,
		[If, [IGet, Transform:Edit] == 0,
			[Note, "Reference Scaler: put the model in Edit mode first."]
			,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[If, zrsSx <= 0,
				[Note, "Reference Scaler: could not read the model scale."]
				,
				[VarSet, zrsBaseScale, zrsSx]
				[VarSet, zrsBaseSize, [IGet, "Texture:Image Plane:Image Size"]]
				[VarSet, zrsAnchored, 1]
			]
		]
	]
	[RoutineCall, ZRSSave]
]

// Sets Image Size to baseSize * (current scale / anchored scale) and leaves a
// one-line result in zrsMsg for the caller to show.
[RoutineDef, ZRSUpdate,
	[VarSet, zrsBefore, 0]
	[VarSet, zrsTarget, 0]
	[If, zrsAnchored == 0,
		[VarSet, zrsMsg, [StrMerge, "Reference Scaler ", zrsBuild, ": not anchored"]]
		,
		[If, [IGet, Transform:Edit] == 0,
			[VarSet, zrsMsg, [StrMerge, "Reference Scaler ", zrsBuild, ": model not in Edit mode"]]
			,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsRatio, zrsSx / zrsBaseScale]
			[VarSet, zrsTarget, zrsBaseSize * zrsRatio]
			[If, zrsTarget < [IGetMin, "Texture:Image Plane:Image Size"], [VarSet, zrsTarget, [IGetMin, "Texture:Image Plane:Image Size"]]]
			[If, zrsTarget > [IGetMax, "Texture:Image Plane:Image Size"], [VarSet, zrsTarget, [IGetMax, "Texture:Image Plane:Image Size"]]]
			[VarSet, zrsBefore, [IGet, "Texture:Image Plane:Image Size"]]
			[ISet, "Texture:Image Plane:Image Size", zrsTarget]
			[VarSet, zrsMsg, [StrMerge, "Reference Scaler ", zrsBuild, ": scale ", zrsSx, " / ", zrsBaseScale]]
			[VarSet, zrsMsg, [StrMerge, zrsMsg, "  Image Size ", zrsBefore, " -> ", [IGet, "Texture:Image Plane:Image Size"]]]
		]
	]
]

// Live loop: wakes on the timer (1) or on left mouse button up (8), i.e. when
// a zoom drag finishes, updates, then goes back to sleep. Another ZScript
// starting ends the loop (ZBrush only runs one script at a time).
[RoutineDef, ZRSLiveLoop,
	[Sleep, 0.5,
		[If, zrsLive,
			[RoutineCall, ZRSUpdate]
			[NoteBar, zrsMsg]
			[SleepAgain, 0.5, 9]
		]
	, 9]
]

// ---------------------------------------------------------------------------
// Interface
// ---------------------------------------------------------------------------
// Each button writes a note-bar message before and after every step, so if it
// stops, the last message shows where.
[ISubPalette, "ZPlugin:Reference Scaler b15"]

[IButton, "ZPlugin:Reference Scaler b15:Anchor",
	"Line the Image Plane reference up with the model, then press to record this as the 1:1 zoom.",
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Anchor started"]]
	[RoutineCall, ZRSAnchor]
	[If, zrsAnchored,
		[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": anchored at scale ", zrsBaseScale, ", Image Size ", zrsBaseSize]]
		,
		[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Anchor failed"]]
	]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b15:Sync",
	"Rescale the reference to the current zoom once.",
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Sync started"]]
	[RoutineCall, ZRSLoad]
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Sync loaded anchor ", zrsAnchored, " scale ", zrsBaseScale]]
	[RoutineCall, ZRSUpdate]
	[NoteBar, zrsMsg]
	, 0, 0.5
]

[ISwitch, "ZPlugin:Reference Scaler b15:Live", 0,
	"Keep rescaling the reference as the camera zooms.",
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Live started"]]
	[RoutineCall, ZRSLoad]
	[If, zrsAnchored == 0, [RoutineCall, ZRSAnchor]]
	[If, zrsAnchored,
		[VarSet, zrsLive, 1]
		[RoutineCall, ZRSUpdate]
		[NoteBar, zrsMsg]
		[RoutineCall, ZRSLiveLoop]
		,
		[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Live needs Anchor"]]
		[ISet, "ZPlugin:Reference Scaler b15:Live", 0]
	]
	,
	[VarSet, zrsLive, 0]
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Live off"]]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b15:Reset",
	"Put Image Size back to its anchored value.",
	[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Reset started"]]
	[RoutineCall, ZRSLoad]
	[If, zrsAnchored,
		[ISet, "Texture:Image Plane:Image Size", zrsBaseSize]
		[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Image Size reset to ", zrsBaseSize]]
		,
		[NoteBar, [StrMerge, "Reference Scaler ", zrsBuild, ": Reset needs Anchor"]]
	]
	, 0, 0.5
]

[IButton, "ZPlugin:Reference Scaler b15:About",
	"Show which build of the script this button is running.",
	[Note, [StrMerge, "Reference Scaler ", zrsBuild]]
	, 0, 0.5
]
