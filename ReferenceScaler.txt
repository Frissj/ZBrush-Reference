// ReferenceScaler.txt
// ZBrush 2022 ZScript plugin.
// Shows a reference image behind the model being edited and keeps it scaled
// and positioned with the model as the camera zooms and pans.
//
// Image Plane only applies its Image Size when an image is loaded, so this
// plugin draws the background itself: ReferenceScaler64.dll renders the image,
// scaled and placed, into a document-sized BMP, which is imported as a texture
// and painted in with Color > FillLayer (with the model in Edit mode, FillLayer
// paints behind it, and X-ray works).
//
// Install: keep the ReferenceScaler_Data folder next to this file, then
// ZScript > Load > pick this file. The buttons appear under
// ZPlugin > Reference Scaler b<build>. ZBrush keeps buttons bound to the
// version that first created them until it restarts, so the sub-palette name
// carries the build number: each new build gets fresh buttons without a restart.
//
// ZScript file commands need "!:" in front of an absolute path; without it
// Texture > Import reports "Unable to open specified file".
//
// Feedback uses [Note, text] popups only: the note bar is cleared when the
// script ends. The anchor is stored in sliders rather than variables, because
// variables reset whenever ZBrush reloads the script.

[IConfig, 2022]
[IShowActions, 0]

[VarDef, zrsDll, "ReferenceScaler_Data\ReferenceScaler64.dll"]

[VarDef, zrsAy, 0]

// Renders a frame with the DLL for zrsRatio, model position (zrsPx, zrsPy)
// and anchor position (zrsAx, zrsAy), imports it as a texture and fills the
// canvas behind the model with it. Frame names alternate so ZBrush never
// reuses a cached texture. Leaves a result in zrsMsg; zrsResult is 1 on
// success. Ratio 1 with model position = anchor position draws the base
// layout: the image fitted to the document. Texture > Clear Depth is turned on
// before the fill so it goes behind the model rather than in front of it, and
// the model's transform is set back to itself afterwards to make ZBrush redraw
// the model over the new background.
[RoutineDef, ZRSDraw,
	[VarSet, zrsResult, 0]
	[If, [IGet, Transform:Edit] == 0,
		[VarSet, zrsMsg, "Put the model in Edit mode first (FillLayer would paint over it otherwise)."]
		,
		[VarSet, zrsFrame, 1 - zrsFrame]
		[VarSet, zrsFrameName, [StrMerge, "frame", zrsFrame, ".png"]]
		[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " ", zrsRatio, " ", zrsPx]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " ", zrsPy, " ", zrsAx, " ", zrsAy]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " ", zrsFrameName]]
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSRender", zrsArgs]]
		[If, zrsResult != 1,
			[VarSet, zrsMsg, [StrMerge, "Render failed (", zrsResult, "). -3 means no image: press Load Image.\nArgs: ", zrsArgs]]
			,
			[VarSet, zrsFile, [FileNameResolvePath, [StrMerge, "ReferenceScaler_Data\frame", zrsFrame, ".png"]]]
			[RoutineCall, ZRSImport]
			[VarSet, zrsDepth, -1]
			[If, [IExists, "Texture:Clear Depth"],
				[ISet, "Texture:Clear Depth", 1]
				[VarSet, zrsDepth, [IGet, "Texture:Clear Depth"]]
			]
			[IPress, Color:FillLayer]
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[TransformSet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsLastRatio, zrsRatio]
			[VarSet, zrsLastPx, zrsPx]
			[VarSet, zrsLastPy, zrsPy]
			[VarSet, zrsMsg, [StrMerge, "Drew ", zrsFrameName, ", ratio ", zrsRatio]]
			[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nFrame exists: ", [FileExists, zrsFile], "  Clear Depth: ", zrsDepth]]
			[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nTexture before ", zrsTexBefore, ", last ", zrsTexMax, ", now ", zrsTexAfter]]
		]
	]
]

// Works out the ratio and positions from the current model and the anchor
// sliders, then draws. Leaves a result in zrsMsg.
[RoutineDef, ZRSUpdate,
	[If, [IGet, "ZPlugin:Reference Scaler b28:Anchor Scale"] <= 0,
		[VarSet, zrsMsg, "Not anchored: press Anchor first."]
		[VarSet, zrsResult, 0]
		,
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsRatio, zrsSx / [IGet, "ZPlugin:Reference Scaler b28:Anchor Scale"]]
		[VarSet, zrsAx, [IGet, "ZPlugin:Reference Scaler b28:Anchor X"]]
		[VarSet, zrsAy, [IGet, "ZPlugin:Reference Scaler b28:Anchor Y"]]
		[RoutineCall, ZRSDraw]
	]
]

// Live loop: wakes on the timer (1) or on left mouse button up (8), i.e. when
// a zoom drag finishes, and redraws only if the zoom or position changed. No
// popups, since they would block zooming. Another ZScript starting ends the
// loop (ZBrush only runs one script at a time).
[RoutineDef, ZRSLiveLoop,
	[Sleep, 0.5,
		[If, zrsLive,
			[If, [IGet, Transform:Edit],
				[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
				[VarSet, zrsRatio, zrsSx / [IGet, "ZPlugin:Reference Scaler b28:Anchor Scale"]]
				[If, ABS(zrsRatio - zrsLastRatio) > 0.002 || ABS(zrsPx - zrsLastPx) > 0.5 || ABS(zrsPy - zrsLastPy) > 0.5,
					[RoutineCall, ZRSUpdate]
				]
			]
			[SleepAgain, 0.5, 9]
		]
	, 9]
]

// Imports zrsFile into the Texture palette and selects it. An import adds the
// texture to the palette but does not make it current, so the last texture
// (the one just imported) is selected through Texture > Item Info. Records
// the index before, the highest index and the index after in zrsTex*.
[RoutineDef, ZRSImport,
	[VarSet, zrsTexBefore, [IGet, "Texture:Item Info"]]
	[FileNameSetNext, [StrMerge, "!:", zrsFile]]
	[IPress, Texture:Import]
	[VarSet, zrsTexMax, [IGetMax, "Texture:Item Info"]]
	[ISet, "Texture:Item Info", zrsTexMax]
	[VarSet, zrsTexAfter, [IGet, "Texture:Item Info"]]
]
