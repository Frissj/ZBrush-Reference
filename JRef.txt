// JRef.txt
// ZBrush 2023 ZScript plugin: keeps reference images with the model.
// A click-through overlay window (drawn by JRef_Data\JRef64.dll) shows a
// reference image per view behind the model (Classic) or in a panel at the
// side (PureRef), following ZBrush's zoom, pan and view snaps, which the DLL
// reads from ZBrush's memory.
//
// Install: put JRef.zsc (ZBrush writes it next to this file when this file is
// loaded with ZScript > Load) and the JRef_Data folder into ZBrush's
// ZStartup\ZPlugs64 folder; JRef then loads every time ZBrush starts, as the
// JRef menu. Where JRef_Data cannot be written, the DLL writes its frames,
// log and settings to C:\Users\Public\Documents\JRef instead.
//
// A project's views, outlines and settings live in a JRef folder next to the
// first image loaded, with copies of the images, so the folder can be taken to
// another computer and opened with Project > Open Project.
//
// ZScript file commands need "!:" in front of an absolute path. Feedback uses
// [Note, text] popups only.

// Sets the variables for the base layout: the image fitted to the document,
// unscaled (ratio 1, model position = anchor position).
// Warns if the DLL ZBrush has loaded is not the one this build expects:
// once the overlay has run, ZBrush keeps the old DLL until it restarts.
[RoutineDef, ZRSCheckDll,
	[VarSet, zrsTick, [FileExecute, zrsDll, "RSVersion"]]
	[If, zrsTick != zrsDllBuild,
		[Note, [StrMerge, "Wrong helper DLL: it is build ", zrsTick, ", this plugin needs ", zrsDllBuild, ".\nRestart ZBrush, and make sure this file is the DLL being updated:\n", [FileNameResolvePath, zrsDll]]]
	]
]

[RoutineDef, ZRSBase,

// FillLayer drops the model out of Edit mode (it is baked into the canvas and
// painted over). ZRSRestore puts it back: if Edit is off after the fill, it
// draws the current tool at the model's saved position with a canvas drag (in
// Draw mode, which ZBrush returns to when Edit turns off; Transform:Draw is
// pressed only where that path exists),
// turns Edit on and restores the transform saved before the fill (zrsK*).
[RoutineDef, ZRSRestore,
	[VarSet, zrsEditFill, [IGet, Transform:Edit]]
	[If, zrsEditFill == 0,
		[If, [IExists, "Transform:Draw"], [IPress, Transform:Draw]]
		[CanvasClick, zrsKx, zrsKy, zrsKx + 40, zrsKy + 40]
		[IPress, Transform:Edit]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	]
	[VarSet, zrsEditAfter, [IGet, Transform:Edit]]
]

// Clears the canvas layer, imports the frame at zrsFile, selects it and fills
// the canvas with it. The fill covers the model until ZRSRestore draws it
// again. Without the clear, FillLayer keeps the depth of the model it baked
// into the canvas, and the redrawn model fights with that copy (stripes of
// fill colour on the model, and clipping when it is smaller than the copy).
[RoutineDef, ZRSFillOnly,
	[VarSet, zrsCleared, [IExists, "Layer:Clear"]]
	[If, zrsCleared, [IPress, "Layer:Clear"]]
	[FileNameSetNext, [StrMerge, "!:", zrsFile]]
	[IPress, Texture:Import]
	[ISet, "Texture:Item Info", [IGetMax, "Texture:Item Info"]]
	[IPress, Color:FillLayer]
]

// Fills the canvas with the frame at zrsFile and puts the model back in Edit
// mode in front of it.
[RoutineDef, ZRSFill,
	[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSFillOnly]
	[RoutineCall, ZRSRestore]
]

// Sets zrsFile to frame0.png where the DLL writes it: JRef_Data, or
// C:/Users/Public/Documents/JRef when JRef_Data cannot be written (JRef
// installed in ZBrush's ZStartup folder without admin rights). nav_log.txt
// and the snapshots are there too.
[RoutineDef, ZRSFramePath,
	[If, [FileExecute, zrsDll, "RSDataFolder"] == 1,
		[VarSet, zrsFile, "C:/Users/Public/Documents/JRef/frame0.png"]
		,
		[VarSet, zrsFile, [FileNameResolvePath, "JRef_Data\frame0.png"]]
	]
]

// Overlay (stage 1 of the live reference): a click-through window drawn by
// the DLL over the canvas. The DLL finds the canvas on screen itself, by
// matching the calibration pattern (which the canvas must be showing) against
// a capture of ZBrush's window. It starts with the last rendered frame's zoom
// and position.
[RoutineDef, ZRSOverlayOn,
	[IUpdate, 2, 1]
	[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " 255"]]
	[RoutineCall, ZRSLevels]
	[VarSet, zrsResult, [FileExecute, zrsDll, "RSOverlayStart", zrsArgs]]
	[VarSet, zrsOx, [FileExecute, zrsDll, "RSOverlayInfo", "0"]]
	[VarSet, zrsOy, [FileExecute, zrsDll, "RSOverlayInfo", "1"]]
	[VarSet, zrsZoom, [FileExecute, zrsDll, "RSOverlayInfo", "2"] / 1000]
	[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "3"]]
]

// Sets zrsMsg2 to the name of view zrsTick (slots 0-5).
[RoutineDef, ZRSViewName,
	[VarSet, zrsMsg2, "Front"]
	[If, zrsTick == 1, [VarSet, zrsMsg2, "Back"]]
	[If, zrsTick == 2, [VarSet, zrsMsg2, "Right"]]
	[If, zrsTick == 3, [VarSet, zrsMsg2, "Left"]]
	[If, zrsTick == 4, [VarSet, zrsMsg2, "Top"]]
	[If, zrsTick == 5, [VarSet, zrsMsg2, "Bottom"]]
]

// Loads an image for view zrsView (named zrsMsg2). The model must be in
// Edit mode, the overlay on, and the camera snapped to a view (every angle a
// multiple of 90) that no other view has taken. The current zoom and
// position become the view's anchor, and the model's outline is recorded.
[RoutineDef, ZRSViewLoad,
	[RoutineCall, ZRSCheckDll]
	[VarSet, zrsOk, 1]
	[If, [IGet, Transform:Edit] == 0,
		[VarSet, zrsOk, 0]
		[Note, "Put the model in Edit mode first."]
	]
	[If, zrsOk,
		[If, [FileExecute, zrsDll, "RSOverlayInfo", "4"] != 1,
			[VarSet, zrsOk, 0]
			[Note, "Press Overlay first: the views are shown by the overlay."]
		]
	]
	[If, zrsOk,
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsRot, [StrMerge, zrsRx, " ", zrsRy, " ", zrsRz]]
	]
	// The DLL knows each view's rotation, so an image can be loaded for any
	// view from anywhere; it shows when that view is snapped to.
	[If, zrsOk,
		[VarSet, zrsFile, [FileNameAsk, "Image (*.jpg;*.jpeg;*.png;*.bmp;*.tif;*.psd)|*.jpg;*.jpeg;*.png;*.bmp;*.tif;*.psd||", , [StrMerge, "Choose the ", zrsMsg2, " image"]]]
		[If, [StrLength, zrsFile] == 0,
			[VarSet, zrsOk, 0]
			[Note, "No image chosen."]
		]
	]
	[If, zrsOk,
		[VarSet, zrsArgs, [StrMerge, zrsView, " ", zrsSx, " ", zrsPx, " ", zrsPy, " ", zrsRot]]
		[FileExecute, zrsDll, "RSViewPrep", zrsArgs]
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSViewImage", zrsFile]]
		[If, zrsResult == 2,
			[If, [FileExecute, zrsDll, "RSOverlayInfo", "39"] > 0,
				[Note, [StrMerge, zrsMsg2, " image loaded. It shows when you snap to the ", zrsMsg2, " view."], , 2]
				,
				[Note, [StrMerge, zrsMsg2, " image loaded, but ZBrush's rotation was not found, so views cannot switch.\nPress Overlay again and send nav_log.txt if this keeps happening."]]
			]
		]
		[If, zrsResult == 1,
			[Note, [StrMerge, zrsMsg2, " image loaded (kept with the project in the JRef folder next to the first image).\nLine it up with Adjust Image if it needs it."], , 2]
		]
		[If, zrsResult < 1,
			[Note, [StrMerge, "Could not read the image (", zrsResult, "):\n", zrsFile]]
		]
	]
]

// Removes view zrsView's image (named zrsMsg2).
[RoutineDef, ZRSViewRemove,
	[If, [FileExecute, zrsDll, "RSViewUsed", [StrMerge, zrsView, ""]],
		[FileExecute, zrsDll, "RSViewClear", [StrMerge, zrsView, ""]]
		[Note, [StrMerge, zrsMsg2, " image removed."], , 1]
		,
		[Note, [StrMerge, zrsMsg2, " has no image to remove."], , 1]
	]
]

// The key colour as the document background (Document > Back, gradient Range
// 0) on a cleared canvas: unlike filled pixels it has no depth, so the model
// can never go behind it. The user's background is saved by the DLL and put
// back by Overlay Off.
[RoutineDef, ZRSKeyBackground,
	[If, [IExists, "Document:Back"],
		[VarSet, zrsTick, [IGet, "Document:Back"]]
		[If, zrsTick != 16711935,
			[FileExecute, zrsDll, "RSValue", [StrMerge, "back ", zrsTick]]
			[If, [IExists, "Document:Range"], [FileExecute, zrsDll, "RSValue", [StrMerge, "range ", [IGet, "Document:Range"]]]]
		]
		[IColorSet, 255, 0, 255]
		[IPress, "Document:Back"]
		[If, [IExists, "Document:Range"], [ISet, "Document:Range", 0]]
		[IColorSet, 255, 255, 255]
	]
	[VarSet, zrsCleared, [IExists, "Layer:Clear"]]
	[If, zrsCleared, [IPress, "Layer:Clear"]]
]

[RoutineDef, ZRSRestoreBackground,
	[VarSet, zrsTick, [FileExecute, zrsDll, "RSValue", "back"]]
	[VarSet, zrsOk, [IExists, "Document:Back"]]
	[If, zrsTick <= 0, [VarSet, zrsOk, 0]]
	[If, zrsTick >= 256, [VarSet, zrsOk, 0]]
	[If, zrsOk,
		[IColorSet, zrsTick, zrsTick, zrsTick]
		[IPress, "Document:Back"]
		[IColorSet, 255, 255, 255]
	]
	[If, [IExists, "Document:Range"], [ISet, "Document:Range", [FileExecute, zrsDll, "RSValue", "range"]]]
]

// Finds where ZBrush keeps the model's view in memory, so the DLL can follow
// zooms and pans exactly: reports the current view, sets three test views
// (moved and scaled a little) and a rotated one (which rules out rotation
// matrix entries that equal the scale in the current view) and reports each,
// then puts the view back and reports that. zrsResult is 1 when the DLL has locked on.
[RoutineDef, ZRSMemProbe,
	[VarSet, zrsResult, 0]
	[If, [FileExecute, zrsDll, "RSAnchorGet", "0"] > 0,
		[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
		[VarSet, zrsArgs, [StrMerge, "start ", zrsKsx, " ", zrsKx, " ", zrsKy, " ", [FileExecute, zrsDll, "RSAnchorGet", "0"]]]
		[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		[Loop, 3,
			[VarSet, zrsTick, zrsTick + 1]
			[TransformSet, zrsKx + 11 * zrsTick, zrsKy + 7 * zrsTick, zrsKz, zrsKsx * (1 + 0.04 * zrsTick), zrsKsy * (1 + 0.04 * zrsTick), zrsKsz * (1 + 0.04 * zrsTick), zrsKrx, zrsKry, zrsKrz]
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsArgs, [StrMerge, "next ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [FileExecute, zrsDll, "RSAnchorGet", "0"]]]
			[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx + 15, zrsKry + 25, zrsKrz]
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[VarSet, zrsArgs, [StrMerge, "next ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [FileExecute, zrsDll, "RSAnchorGet", "0"]]]
		[FileExecute, zrsDll, "RSMemProbe", zrsArgs]
		// The rotation too, so the DLL sees snaps to a view.
		[FileExecute, zrsDll, "RSRotProbe", [StrMerge, "start ", zrsRx, " ", zrsRy, " ", zrsRz]]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx + 40, zrsKry - 30, zrsKrz + 10]
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[FileExecute, zrsDll, "RSRotProbe", [StrMerge, "next ", zrsRx, " ", zrsRy, " ", zrsRz]]
		[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
		[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
		[FileExecute, zrsDll, "RSRotProbe", [StrMerge, "last ", zrsRx, " ", zrsRy, " ", zrsRz]]
		[VarSet, zrsArgs, [StrMerge, "last ", zrsSx, " ", zrsPx, " ", zrsPy, " ", [FileExecute, zrsDll, "RSAnchorGet", "0"]]]
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSMemProbe", zrsArgs]]
	]
]

// Logs ZBrush's canvas depth (PixolPick component 1) and colour (0) on a 5 x 5
// grid over the document, labelled with zrsMsg2, to nav_log.txt: shows
// whether empty canvas has depth left in it that could hide the model.
[RoutineDef, ZRSDepthProbe,
	[VarSet, zrsMult, [IGet, Document:Height] * 0.05]
	[Loop, 5,
		[VarSet, zrsTry, [IGet, Document:Width] * 0.05]
		[Loop, 5,
			[VarSet, zrsArgs, [StrMerge, zrsMsg2, " at ", zrsTry, " ", zrsMult, " depth "]]
			[FileExecute, zrsDll, "RSLog", [StrMerge, zrsArgs, [PixolPick, 1, zrsTry, zrsMult], " colour ", [PixolPick, 0, zrsTry, zrsMult]]]
			[VarSet, zrsTry, zrsTry + [IGet, Document:Width] * 0.225]
		]
		[VarSet, zrsMult, zrsMult + [IGet, Document:Height] * 0.225]
	]
]

// Counts in zrsCount how many points of a 30 x 30 grid over the document
// ZBrush has drawn something other than the key colour at (the model).
[RoutineDef, ZRSCountModel,
	[VarSet, zrsCount, 0]
	[VarSet, zrsMult, [IGet, Document:Height] / 60]
	[Loop, 30,
		[VarSet, zrsTry, [IGet, Document:Width] / 60]
		[Loop, 30,
			[If, [PixolPick, 0, zrsTry, zrsMult] != 16711935, [VarSet, zrsCount, zrsCount + 1]]
			[VarSet, zrsTry, zrsTry + [IGet, Document:Width] / 30]
		]
		[VarSet, zrsMult, zrsMult + [IGet, Document:Height] / 30]
	]
]

// ZBrush stops drawing what is behind its back clipping plane, and a model
// at the canvas plane (depth 0) is cut off when zoomed out. The ZBrush fix is
// to bring the model forward in depth. This shrinks the view to 30% (where
// the cut-off shows), counts how much of the model ZBrush draws with it 5x
// its scale in front of and behind its depth, keeps whichever shows more,
// and restores the zoom. The factor is kept with RSValue "depth".
[RoutineDef, ZRSDepthFix,
	[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[VarSet, zrsBestZ, zrsKz]
	[TransformSet, zrsKx, zrsKy, zrsKz, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[VarSet, zrsBestCount, zrsCount]
	[VarSet, zrsMemFound, zrsCount]
	[TransformSet, zrsKx, zrsKy, zrsKz - 5 * zrsKsx, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[VarSet, zrsTurn, zrsCount]
	[If, zrsCount > zrsBestCount + 2,
		[VarSet, zrsBestZ, zrsKz - 5 * zrsKsx]
	]
	[If, zrsCount > zrsBestCount + 2,
		[VarSet, zrsBestCount, zrsCount]
	]
	[TransformSet, zrsKx, zrsKy, zrsKz + 5 * zrsKsx, zrsKsx * 0.3, zrsKsy * 0.3, zrsKsz * 0.3, zrsKrx, zrsKry, zrsKrz]
	[RoutineCall, ZRSCountModel]
	[If, zrsCount > zrsBestCount + 2,
		[VarSet, zrsBestZ, zrsKz + 5 * zrsKsx]
	]
	[If, zrsCount > zrsBestCount + 2,
		[VarSet, zrsBestCount, zrsCount]
	]
	[FileExecute, zrsDll, "RSLog", [StrMerge, "document ", [IGet, Document:Width], " x ", [IGet, Document:Height]]]
	[VarSet, zrsArgs, [StrMerge, "depth fix: z ", zrsKz, " shows ", zrsMemFound, ", z-5s shows ", zrsTurn, ", z+5s shows ", zrsCount]]
	[FileExecute, zrsDll, "RSLog", [StrMerge, zrsArgs, ", using ", zrsBestZ]]
	[TransformSet, zrsKx, zrsKy, zrsBestZ, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
	[If, zrsBestZ != zrsKz,
		[FileExecute, zrsDll, "RSValue", [StrMerge, "depth ", zrsBestZ / zrsKsx]]
		,
		[FileExecute, zrsDll, "RSValue", "depth 0"]
	]
]

// Overlay: works out the layout for the current camera (rendering it to
// frame0.png also saves the layout for the DLL), fills the canvas with the
// calibration pattern, which also hides the model, waits a second so ZBrush
// repaints the screen, lets the DLL find the pattern on screen, then fills the
// canvas with the key colour magenta (or with the reference if that failed)
// and puts the model back. The overlay shows the reference wherever it sees
// magenta, so the model stays in front.
// Every fill uses frame0.png: importing a name that already exists replaces
// that texture in place, so a single name keeps it the last texture.
[IButton, "JRef:Main:Overlay",
	"Show the reference in a click-through window over the canvas, behind the model. Press again to find the canvas again.",
	[RoutineCall, ZRSCheckDll]
	[FileExecute, zrsDll, "RSOverlayStop"]
	[If, [IGet, Transform:Edit] == 0,
		[Note, "Put the model in Edit mode first."]
		,
				[VarSet, zrsAutoAnchored, 0]
		[If, [FileExecute, zrsDll, "RSAnchorGet", "0"] <= 0,
			[RoutineCall, ZRSAnchorNow]
			[VarSet, zrsAutoAnchored, 1]
		]
		[If, [FileExecute, zrsDll, "RSAnchorGet", "0"] > 0,
			[TransformGet, zrsPx, zrsPy, zrsPz, zrsSx, zrsSy, zrsSz, zrsRx, zrsRy, zrsRz]
			[VarSet, zrsRatio, zrsSx / [FileExecute, zrsDll, "RSAnchorGet", "0"]]
			[VarSet, zrsAx, [FileExecute, zrsDll, "RSAnchorGet", "1"]]
			[VarSet, zrsAy, [FileExecute, zrsDll, "RSAnchorGet", "2"]]
			,
			[RoutineCall, ZRSBase]
		]
		[VarSet, zrsArgs, [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " ", zrsRatio, " ", zrsPx]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " ", zrsPy, " ", zrsAx, " ", zrsAy]]
		[VarSet, zrsArgs, [StrMerge, zrsArgs, " frame0.png"]]
		// The overlay shows only the views' images, so no image (-3) is fine.
		// ZScript does not evaluate "a != b && c != d" as two comparisons, so
		// conditions are built one step at a time.
		[VarSet, zrsResult, [FileExecute, zrsDll, "RSRender", zrsArgs]]
		[If, zrsResult == -3, [VarSet, zrsResult, 1]]
		[If, zrsResult != 1,
			[Note, [StrMerge, "Render failed (", zrsResult, ")."]]
			,
			[FileExecute, zrsDll, "RSPattern", [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " frame0.png"]]
			[RoutineCall, ZRSFramePath]
			[TransformGet, zrsKx, zrsKy, zrsKz, zrsKsx, zrsKsy, zrsKsz, zrsKrx, zrsKry, zrsKrz]
			[VarSet, zrsMsg2, "before overlay"]
			[RoutineCall, ZRSDepthProbe]
			[RoutineCall, ZRSFillOnly]
			[VarSet, zrsMsg2, "pattern fill"]
			[RoutineCall, ZRSDepthProbe]
			[Sleep, 1, [VarSet, zrsTick, 0]]
			[RoutineCall, ZRSOverlayOn]
			[If, zrsResult > 0,
				[RoutineCall, ZRSKeyBackground]
				,
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSRender", zrsArgs]]
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "3"]]
				[RoutineCall, ZRSFillOnly]
			]
			[RoutineCall, ZRSRestore]
			[VarSet, zrsLastRatio, zrsRatio]
			[VarSet, zrsLastPx, zrsPx]
			[VarSet, zrsLastPy, zrsPy]
			[If, zrsResult > 0,
				[VarSet, zrsArgs, [StrMerge, "Overlay on: found the canvas at ", zrsOx, ", ", zrsOy, ", ", zrsZoom, " screen px per document px (match ", zrsTick, "%)"]]
				[VarSet, zrsResult, [FileExecute, zrsDll, "RSOverlayInfo", "5"]]
				[If, zrsResult == 1,
					[VarSet, zrsMsg, "Model in front: keying on (1)."]
					,
					[VarSet, zrsMsg, [StrMerge, "Keying unavailable (", zrsResult, "): 0 = this Windows cannot exclude the overlay from capture, -1 = exclusion did not work. The overlay covers the model."]]
				]
				[Sleep, 0.5, [VarSet, zrsTick, 0]]
				[VarSet, zrsMsg2, "overlay ready"]
				[RoutineCall, ZRSDepthProbe]
				[VarSet, zrsTick, [FileExecute, zrsDll, "RSOverlayInfo", "32"]]
				[If, zrsTick < 5,
					[FileExecute, zrsDll, "RSBlank", [StrMerge, [IGet, Document:Width], " ", [IGet, Document:Height], " frame0.png 255 0 255"]]
					[RoutineCall, ZRSFramePath]
					[RoutineCall, ZRSFill]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nBackground colour did not turn magenta (", zrsTick, "%): using a fill, the model may clip."]]
				]
				[VarSet, zrsTick, 0]
				[RoutineCall, ZRSMemProbe]
				[VarSet, zrsSetResult, zrsResult]
				[RoutineCall, ZRSDepthFix]
				[VarSet, zrsResult, zrsSetResult]
				[If, zrsResult == 1,
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nFound ZBrush's view in memory: zoom and pan follow exactly."]]
					[If, [FileExecute, zrsDll, "RSOverlayInfo", "39"] > 0,
						[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nFound the rotation too: snapping to a view switches its image."]]
						,
						[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nRotation not found in memory: views will not switch. Please send nav_log.txt."]]
					]

					,
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nZBrush's view not found in memory (", zrsResult, "), candidates ", [FileExecute, zrsDll, "RSOverlayInfo", "23"], " ", [FileExecute, zrsDll, "RSOverlayInfo", "24"], " ", [FileExecute, zrsDll, "RSOverlayInfo", "25"]]]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, "\nScan: ", [FileExecute, zrsDll, "RSOverlayInfo", "27"], " MB, x matches ", [FileExecute, zrsDll, "RSOverlayInfo", "28"], ", pairs ", [FileExecute, zrsDll, "RSOverlayInfo", "29"]]]
					[VarSet, zrsMsg, [StrMerge, zrsMsg, ", scales ", [FileExecute, zrsDll, "RSOverlayInfo", "30"], ", ", [FileExecute, zrsDll, "RSOverlayInfo", "31"], " ms. View ", zrsKsx, " ", zrsKx, " ", zrsKy]]
				]
				[If, zrsCleared == 0, [VarSet, zrsMsg, [StrMerge, zrsMsg, "\nLayer:Clear not found, the model may show stripes."]]]
				[If, zrsAutoAnchored, [VarSet, zrsMsg, [StrMerge, zrsMsg, "\nNo anchor was set: anchored at the current view."]]]
				[Note, [StrMerge, zrsArgs, "\n", zrsMsg]]
				[If, [FileExecute, zrsDll, "RSValue", "depth"] != 0,
					[Note, [StrMerge, "Moved the model off ZBrush's back clipping plane, to depth ", zrsBestZ]]
					,
					[Note, [StrMerge, "Depth unchanged: the model showed at ", zrsMemFound, " / ", zrsTurn, " / ", zrsCount, " of 900 points (now / in front / behind)."]]
				]
				// No image is shown until a view is snapped to (the DLL looks views
				// up as the rotation changes).
				,
				[VarSet, zrsMsg, "-3: frame0.png missing.\n-4: screen capture failed.\n-5: the pattern was not found on screen."]
				[VarSet, zrsArgs, [StrMerge, "Overlay failed (", zrsResult, "). Best match ", zrsTick, "% at ", zrsOx, ", ", zrsOy]]
				[Note, [StrMerge, zrsArgs, " scale ", zrsZoom, "\n", zrsMsg]]
				[RoutineCall, ZRSFramePath]
				[Note, [StrMerge, "What the plugin saw is saved as capture.png next to\n", zrsFile, "\nPlease send that file."]]
			]
		]
	]
	, 0, 0.5
]
